# STATUS QUERY - Balas whisper "@status" dengan info char
# ==========================================================

automacro statusQuery {
#    PrivMsg /^\@status$/i
    timeout 3
    call statusReply
}

macro statusReply {
    $sender = $.PrivMsgLastName
    $blvl   = &eval($char->{lv})
    $jlvl   = &eval($char->{lv_job})
    $hp     = &eval($char->{hp})
    $hpmax  = &eval($char->{hp_max})
    $hppct  = &eval(int($char->{hp} / $char->{hp_max} * 100))
    $sp     = &eval($char->{sp})
    $spmax  = &eval($char->{sp_max})
    $wt     = &eval($char->{weight})
    $wtmax  = &eval($char->{weight_max})
    $wtpct  = &eval(int($char->{weight} / $char->{weight_max} * 100))
    $zeny   = &eval($char->{zeny})
    $map    = &eval($field->baseName)
    $px     = &eval($char->{pos_to}{x})
    $py     = &eval($char->{pos_to}{y})
    $name   = &nick
    do pm $sender [BOT] Char: $name | Lv: $blvl | Job Lv: $jlvl
    do pm $sender HP: $hp/$hpmax ($hppct%) | SP: $sp/$spmax
    do pm $sender Weight: $wt/$wtmax ($wtpct%) | Zeny: $zeny z
    do pm $sender Map: $map ($px, $py)
}
